# TODO 清单功能实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use /feature-dev (recommended) or /plan to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 为 AIoT 墨水屏设备增加 TODO 清单功能，支持手机 WiFi AP 直连上传任务，墨水屏触摸操作。

**Architecture:** 4 个新组件（todo_manager/wifi_ap/todo_web/todo_ui）+ 2 个现有组件改动（ui/main）。数据层 NVS 存储 + 内存缓存，HTTP REST API 供手机访问，LVGL 页面显示。

**Tech Stack:** ESP-IDF v5.5.4, LVGL, cJSON, esp_http_server, NVS

## Global Constraints

- ESP-IDF v5.5.4, 工具链在 `E:\Espressif\tools`
- 墨水屏 SSD1681 单色屏，无法渲染 emoji，用 `[!]/[!!]/[!!!]` 文字标记
- NVS 单个 value 限制约 1900 字节，任务需分块存储
- PSRAM 必须启用，LVGL 缓冲区在 PSRAM
- WiFi AP 与 STA 互斥，AP 开启时暂停 STA
- 所有组件用 C 语言，头文件加 `extern "C"` 保护
- 构建命令：`idf.py build`

### Step 3: 创建 todo_manager.cpp 实现

完整实现，包含：
- NVS namespace `todo` 读写
- 内存缓存 + mutex 保护
- 分块存储（每块 12 条任务）
- ID 生成（清单 l_001, 任务 t_xxxx）
- 变更回调

```cpp
#include "todo_manager.h"
#include <cstring>
#include <cstdio>
#include <nvs_flash.h>
#include <nvs.h>
#include <cJSON.h>
#include <esp_log.h>
#include <esp_timer.h>
#include <freertos/FreeRTOS.h>
#include <freertos/semphr.h>

static const char *TAG = "todo_mgr";
static const char *NVS_NAMESPACE = "todo";
static const char *LISTS_KEY = "lists";
static const int TASKS_PER_CHUNK = 12;

// 内存缓存
static todo_list_t s_lists[TODO_MAX_LISTS];
static uint8_t s_list_count = 0;
static todo_task_t s_tasks[TODO_MAX_LISTS][TODO_MAX_TASKS];
static uint16_t s_task_counts[TODO_MAX_LISTS] = {0};

static SemaphoreHandle_t s_mutex = NULL;
static todo_change_cb_t s_change_cb = NULL;

// NVS handle
static nvs_handle_t s_nvs = 0;

// 内部函数声明
static esp_err_t load_lists_from_nvs(void);
static esp_err_t save_lists_to_nvs(void);
static esp_err_t load_tasks_from_nvs(uint8_t list_idx);
static esp_err_t save_tasks_to_nvs(uint8_t list_idx);
static int find_list_idx(const char *id);
static int find_task_list_idx(const char *task_id, int *task_idx_out);
static void notify_change(void);

esp_err_t TodoMgr_Init(void)
{
    s_mutex = xSemaphoreCreateMutex();
    if (!s_mutex) {
        ESP_LOGE(TAG, "Failed to create mutex");
        return ESP_FAIL;
    }

    esp_err_t err = nvs_open(NVS_NAMESPACE, NVS_READWRITE, &s_nvs);
    if (err != ESP_OK) {
        ESP_LOGE(TAG, "NVS open failed: %s", esp_err_to_name(err));
        return err;
    }

    err = load_lists_from_nvs();
    if (err != ESP_OK) {
        ESP_LOGW(TAG, "No existing lists, starting fresh");
        s_list_count = 0;
    }

    ESP_LOGI(TAG, "Init done, %d lists loaded", s_list_count);
    return ESP_OK;
}

esp_err_t TodoMgr_GetLists(todo_list_t *lists, uint8_t *count)
{
    if (!lists || !count) return ESP_ERR_INVALID_ARG;

    xSemaphoreTake(s_mutex, portMAX_DELAY);
    *count = s_list_count;
    memcpy(lists, s_lists, s_list_count * sizeof(todo_list_t));
    xSemaphoreGive(s_mutex);

    return ESP_OK;
}

esp_err_t TodoMgr_CreateList(const char *name, const char *icon, char *out_id)
{
    if (!name || !out_id) return ESP_ERR_INVALID_ARG;

    xSemaphoreTake(s_mutex, portMAX_DELAY);

    if (s_list_count >= TODO_MAX_LISTS) {
        xSemaphoreGive(s_mutex);
        ESP_LOGW(TAG, "Max lists reached");
        return ESP_ERR_NO_MEM;
    }

    // 生成 ID: l_001, l_002, ...
    int idx = s_list_count;
    snprintf(s_lists[idx].id, TODO_ID_LEN, "l_%03d", idx + 1);
    strncpy(s_lists[idx].name, name, TODO_NAME_LEN - 1);
    s_lists[idx].name[TODO_NAME_LEN - 1] = '\0';

    if (icon) {
        strncpy(s_lists[idx].icon, icon, TODO_ICON_LEN - 1);
        s_lists[idx].icon[TODO_ICON_LEN - 1] = '\0';
    } else {
        strcpy(s_lists[idx].icon, "[]");
    }

    s_lists[idx].order = idx;
    s_lists[idx].task_count = 0;
    s_task_counts[idx] = 0;

    s_list_count++;

    esp_err_t err = save_lists_to_nvs();
    if (err != ESP_OK) {
        ESP_LOGE(TAG, "Save lists failed");
        s_list_count--;
        xSemaphoreGive(s_mutex);
        return err;
    }

    strcpy(out_id, s_lists[idx].id);
    xSemaphoreGive(s_mutex);

    notify_change();
    ESP_LOGI(TAG, "Created list: %s (%s)", out_id, name);
    return ESP_OK;
}

esp_err_t TodoMgr_UpdateList(const char *id, const char *name, const char *icon, const uint8_t *order)
{
    if (!id) return ESP_ERR_INVALID_ARG;

    xSemaphoreTake(s_mutex, portMAX_DELAY);

    int idx = find_list_idx(id);
    if (idx < 0) {
        xSemaphoreGive(s_mutex);
        return ESP_ERR_NOT_FOUND;
    }

    if (name) {
        strncpy(s_lists[idx].name, name, TODO_NAME_LEN - 1);
        s_lists[idx].name[TODO_NAME_LEN - 1] = '\0';
    }
    if (icon) {
        strncpy(s_lists[idx].icon, icon, TODO_ICON_LEN - 1);
        s_lists[idx].icon[TODO_ICON_LEN - 1] = '\0';
    }
    if (order) {
        s_lists[idx].order = *order;
    }

    esp_err_t err = save_lists_to_nvs();
    xSemaphoreGive(s_mutex);

    if (err == ESP_OK) notify_change();
    return err;
}

esp_err_t TodoMgr_DeleteList(const char *id, uint16_t *deleted_tasks)
{
    if (!id) return ESP_ERR_INVALID_ARG;

    xSemaphoreTake(s_mutex, portMAX_DELAY);

    int idx = find_list_idx(id);
    if (idx < 0) {
        xSemaphoreGive(s_mutex);
        return ESP_ERR_NOT_FOUND;
    }

    if (deleted_tasks) *deleted_tasks = s_task_counts[idx];

    // 删除 NVS 中的任务块
    char key[16];
    snprintf(key, sizeof(key), "%s_count", s_lists[idx].id);
    nvs_erase_key(s_nvs, key);

    int chunks = (s_task_counts[idx] + TASKS_PER_CHUNK - 1) / TASKS_PER_CHUNK;
    for (int i = 0; i < chunks; i++) {
        snprintf(key, sizeof(key), "%s_%d", s_lists[idx].id, i);
        nvs_erase_key(s_nvs, key);
    }

    // 移动后续列表
    for (int i = idx; i < s_list_count - 1; i++) {
        s_lists[i] = s_lists[i + 1];
        s_task_counts[i] = s_task_counts[i + 1];
        memcpy(s_tasks[i], s_tasks[i + 1], sizeof(s_tasks[0]));
    }
    s_list_count--;

    esp_err_t err = save_lists_to_nvs();
    nvs_commit(s_nvs);
    xSemaphoreGive(s_mutex);

    if (err == ESP_OK) notify_change();
    ESP_LOGI(TAG, "Deleted list: %s (%d tasks)", id, deleted_tasks ? *deleted_tasks : 0);
    return err;
}

esp_err_t TodoMgr_GetTasks(const char *list_id, todo_task_t *tasks, uint16_t *count)
{
    if (!list_id || !tasks || !count) return ESP_ERR_INVALID_ARG;

    xSemaphoreTake(s_mutex, portMAX_DELAY);

    int idx = find_list_idx(list_id);
    if (idx < 0) {
        xSemaphoreGive(s_mutex);
        return ESP_ERR_NOT_FOUND;
    }

    *count = s_task_counts[idx];
    memcpy(tasks, s_tasks[idx], s_task_counts[idx] * sizeof(todo_task_t));
    xSemaphoreGive(s_mutex);

    return ESP_OK;
}

esp_err_t TodoMgr_CreateTask(const char *list_id, const char *text, uint8_t priority, const char *due, char *out_id)
{
    if (!list_id || !text || !out_id) return ESP_ERR_INVALID_ARG;

    xSemaphoreTake(s_mutex, portMAX_DELAY);

    int idx = find_list_idx(list_id);
    if (idx < 0) {
        xSemaphoreGive(s_mutex);
        return ESP_ERR_NOT_FOUND;
    }

    if (s_task_counts[idx] >= TODO_MAX_TASKS) {
        xSemaphoreGive(s_mutex);
        ESP_LOGW(TAG, "Max tasks reached for list %s", list_id);
        return ESP_ERR_NO_MEM;
    }

    int task_idx = s_task_counts[idx];
    todo_task_t *task = &s_tasks[idx][task_idx];

    // 生成 ID: t_xxxx (时间戳低4位)
    uint32_t ts = (uint32_t)(esp_timer_get_time() / 1000000);
    snprintf(task->id, TODO_ID_LEN, "t_%04u", ts % 10000);
    strncpy(task->text, text, TODO_TEXT_LEN - 1);
    task->text[TODO_TEXT_LEN - 1] = '\0';
    task->done = false;
    task->priority = priority;
    task->created = ts;

    if (due) {
        strncpy(task->due, due, TODO_DATE_LEN - 1);
        task->due[TODO_DATE_LEN - 1] = '\0';
    } else {
        task->due[0] = '\0';
    }

    s_task_counts[idx]++;
    s_lists[idx].task_count = s_task_counts[idx];

    esp_err_t err = save_tasks_to_nvs(idx);
    if (err != ESP_OK) {
        s_task_counts[idx]--;
        s_lists[idx].task_count = s_task_counts[idx];
        xSemaphoreGive(s_mutex);
        return err;
    }

    save_lists_to_nvs();
    strcpy(out_id, task->id);
    xSemaphoreGive(s_mutex);

    notify_change();
    ESP_LOGI(TAG, "Created task: %s in %s", out_id, list_id);
    return ESP_OK;
}

esp_err_t TodoMgr_UpdateTask(const char *task_id, const bool *done, const uint8_t *priority, const char *due)
{
    if (!task_id) return ESP_ERR_INVALID_ARG;

    xSemaphoreTake(s_mutex, portMAX_DELAY);

    int list_idx, task_idx;
    list_idx = find_task_list_idx(task_id, &task_idx);
    if (list_idx < 0) {
        xSemaphoreGive(s_mutex);
        return ESP_ERR_NOT_FOUND;
    }

    todo_task_t *task = &s_tasks[list_idx][task_idx];
    if (done) task->done = *done;
    if (priority) task->priority = *priority;
    if (due) {
        strncpy(task->due, due, TODO_DATE_LEN - 1);
        task->due[TODO_DATE_LEN - 1] = '\0';
    }

    esp_err_t err = save_tasks_to_nvs(list_idx);
    xSemaphoreGive(s_mutex);

    if (err == ESP_OK) notify_change();
    return err;
}

esp_err_t TodoMgr_DeleteTask(const char *task_id)
{
    if (!task_id) return ESP_ERR_INVALID_ARG;

    xSemaphoreTake(s_mutex, portMAX_DELAY);

    int list_idx, task_idx;
    list_idx = find_task_list_idx(task_id, &task_idx);
    if (list_idx < 0) {
        xSemaphoreGive(s_mutex);
        return ESP_ERR_NOT_FOUND;
    }

    // 移动后续任务
    for (int i = task_idx; i < s_task_counts[list_idx] - 1; i++) {
        s_tasks[list_idx][i] = s_tasks[list_idx][i + 1];
    }
    s_task_counts[list_idx]--;
    s_lists[list_idx].task_count = s_task_counts[list_idx];

    esp_err_t err = save_tasks_to_nvs(list_idx);
    save_lists_to_nvs();
    xSemaphoreGive(s_mutex);

    if (err == ESP_OK) notify_change();
    ESP_LOGI(TAG, "Deleted task: %s", task_id);
    return err;
}

void TodoMgr_SetChangeCallback(todo_change_cb_t cb)
{
    s_change_cb = cb;
}

// 内部函数实现

static esp_err_t load_lists_from_nvs(void)
{
    size_t required_size = 0;
    esp_err_t err = nvs_get_blob(s_nvs, LISTS_KEY, NULL, &required_size);
    if (err != ESP_OK || required_size == 0) return ESP_ERR_NOT_FOUND;

    char *buf = (char *)malloc(required_size);
    if (!buf) return ESP_ERR_NO_MEM;

    err = nvs_get_blob(s_nvs, LISTS_KEY, buf, &required_size);
    if (err != ESP_OK) {
        free(buf);
        return err;
    }

    cJSON *json = cJSON_Parse(buf);
    free(buf);
    if (!json || !cJSON_IsArray(json)) {
        cJSON_Delete(json);
        return ESP_ERR_INVALID_ARG;
    }

    s_list_count = cJSON_GetArraySize(json);
    if (s_list_count > TODO_MAX_LISTS) s_list_count = TODO_MAX_LISTS;

    for (int i = 0; i < s_list_count; i++) {
        cJSON *item = cJSON_GetArrayItem(json, i);
        cJSON *jid = cJSON_GetObjectItem(item, "id");
        cJSON *jname = cJSON_GetObjectItem(item, "name");
        cJSON *jicon = cJSON_GetObjectItem(item, "icon");
        cJSON *jorder = cJSON_GetObjectItem(item, "order");

        if (jid && jid->valuestring)
            strncpy(s_lists[i].id, jid->valuestring, TODO_ID_LEN - 1);
        if (jname && jname->valuestring)
            strncpy(s_lists[i].name, jname->valuestring, TODO_NAME_LEN - 1);
        if (jicon && jicon->valuestring)
            strncpy(s_lists[i].icon, jicon->valuestring, TODO_ICON_LEN - 1);
        if (jorder)
            s_lists[i].order = (uint8_t)jorder->valueint;

        s_lists[i].task_count = 0;
        load_tasks_from_nvs(i);
    }

    cJSON_Delete(json);
    return ESP_OK;
}

static esp_err_t save_lists_to_nvs(void)
{
    cJSON *json = cJSON_CreateArray();
    if (!json) return ESP_ERR_NO_MEM;

    for (int i = 0; i < s_list_count; i++) {
        cJSON *item = cJSON_CreateObject();
        cJSON_AddStringToObject(item, "id", s_lists[i].id);
        cJSON_AddStringToObject(item, "name", s_lists[i].name);
        cJSON_AddStringToObject(item, "icon", s_lists[i].icon);
        cJSON_AddNumberToObject(item, "order", s_lists[i].order);
        cJSON_AddItemToArray(json, item);
    }

    char *str = cJSON_PrintUnformatted(json);
    cJSON_Delete(json);
    if (!str) return ESP_ERR_NO_MEM;

    esp_err_t err = nvs_set_blob(s_nvs, LISTS_KEY, str, strlen(str));
    free(str);

    if (err == ESP_OK) err = nvs_commit(s_nvs);
    return err;
}

static esp_err_t load_tasks_from_nvs(uint8_t list_idx)
{
    char count_key[16];
    snprintf(count_key, sizeof(count_key), "%s_count", s_lists[list_idx].id);

    uint16_t count = 0;
    esp_err_t err = nvs_get_u16(s_nvs, count_key, &count);
    if (err != ESP_OK || count == 0) return ESP_OK;

    if (count > TODO_MAX_TASKS) count = TODO_MAX_TASKS;
    s_task_counts[list_idx] = count;
    s_lists[list_idx].task_count = count;

    int chunks = (count + TASKS_PER_CHUNK - 1) / TASKS_PER_CHUNK;
    int loaded = 0;

    for (int c = 0; c < chunks && loaded < count; c++) {
        char key[16];
        snprintf(key, sizeof(key), "%s_%d", s_lists[list_idx].id, c);

        size_t len = 0;
        err = nvs_get_blob(s_nvs, key, NULL, &len);
        if (err != ESP_OK || len == 0) continue;

        char *buf = (char *)malloc(len);
        if (!buf) continue;

        err = nvs_get_blob(s_nvs, key, buf, &len);
        if (err != ESP_OK) {
            free(buf);
            continue;
        }

        cJSON *arr = cJSON_Parse(buf);
        free(buf);
        if (!arr || !cJSON_IsArray(arr)) continue;

        int arr_size = cJSON_GetArraySize(arr);
        for (int i = 0; i < arr_size && loaded < count; i++) {
            cJSON *item = cJSON_GetArrayItem(arr, i);
            todo_task_t *t = &s_tasks[list_idx][loaded];

            cJSON *jid = cJSON_GetObjectItem(item, "id");
            cJSON *jtext = cJSON_GetObjectItem(item, "text");
            cJSON *jdone = cJSON_GetObjectItem(item, "done");
            cJSON *jpri = cJSON_GetObjectItem(item, "priority");
            cJSON *jdue = cJSON_GetObjectItem(item, "due");
            cJSON *jcreated = cJSON_GetObjectItem(item, "created");

            if (jid) strncpy(t->id, jid->valuestring, TODO_ID_LEN - 1);
            if (jtext) strncpy(t->text, jtext->valuestring, TODO_TEXT_LEN - 1);
            if (jdone) t->done = cJSON_IsTrue(jdone);
            if (jpri) t->priority = (uint8_t)jpri->valueint;
            if (jdue) strncpy(t->due, jdue->valuestring, TODO_DATE_LEN - 1);
            if (jcreated) t->created = (uint32_t)jcreated->valueint;

            loaded++;
        }
        cJSON_Delete(arr);
    }

    return ESP_OK;
}

static esp_err_t save_tasks_to_nvs(uint8_t list_idx)
{
    uint16_t count = s_task_counts[list_idx];
    char count_key[16];
    snprintf(count_key, sizeof(count_key), "%s_count", s_lists[list_idx].id);

    nvs_set_u16(s_nvs, count_key, count);

    int chunks = (count + TASKS_PER_CHUNK - 1) / TASKS_PER_CHUNK;
    int saved = 0;

    for (int c = 0; c < chunks; c++) {
        cJSON *arr = cJSON_CreateArray();
        int end = saved + TASKS_PER_CHUNK;
        if (end > count) end = count;

        for (int i = saved; i < end; i++) {
            todo_task_t *t = &s_tasks[list_idx][i];
            cJSON *item = cJSON_CreateObject();
            cJSON_AddStringToObject(item, "id", t->id);
            cJSON_AddStringToObject(item, "text", t->text);
            cJSON_AddBoolToObject(item, "done", t->done);
            cJSON_AddNumberToObject(item, "priority", t->priority);
            cJSON_AddStringToObject(item, "due", t->due);
            cJSON_AddNumberToObject(item, "created", t->created);
            cJSON_AddItemToArray(arr, item);
        }

        char *str = cJSON_PrintUnformatted(arr);
        cJSON_Delete(arr);

        if (str) {
            char key[16];
            snprintf(key, sizeof(key), "%s_%d", s_lists[list_idx].id, c);
            nvs_set_blob(s_nvs, key, str, strlen(str));
            free(str);
        }
        saved = end;
    }

    return nvs_commit(s_nvs);
}

static int find_list_idx(const char *id)
{
    for (int i = 0; i < s_list_count; i++) {
        if (strcmp(s_lists[i].id, id) == 0) return i;
    }
    return -1;
}

static int find_task_list_idx(const char *task_id, int *task_idx_out)
{
    for (int i = 0; i < s_list_count; i++) {
        for (int j = 0; j < s_task_counts[i]; j++) {
            if (strcmp(s_tasks[i][j].id, task_id) == 0) {
                if (task_idx_out) *task_idx_out = j;
                return i;
            }
        }
    }
    return -1;
}

static void notify_change(void)
{
    if (s_change_cb) s_change_cb();
}
```

```cpp
#include "wifi_ap.h"
#include <cstring>
#include <esp_wifi.h>
#include <esp_event.h>
#include <esp_netif.h>
#include <esp_log.h>
#include <esp_timer.h>
#include <freertos/FreeRTOS.h>
#include <freertos/task.h>

static const char *TAG = "wifi_ap";

static wifi_ap_state_t s_state = WIFI_AP_STATE_IDLE;
static wifi_ap_state_cb_t s_state_cb = NULL;
static esp_timer_handle_t s_timeout_timer = NULL;
static esp_netif_t *s_ap_netif = NULL;
static bool s_initialized = false;

static void timeout_timer_cb(void *arg)
{
    ESP_LOGI(TAG, "AP timeout, stopping...");
    WifiAp_Stop();
}

static void wifi_event_handler(void *arg, esp_event_base_t event_base,
                               int32_t event_id, void *event_data)
{
    if (event_base == WIFI_EVENT) {
        switch (event_id) {
            case WIFI_EVENT_AP_STACONNECTED: {
                wifi_event_ap_staconnected_t *event = (wifi_event_ap_staconnected_t *)event_data;
                ESP_LOGI(TAG, "Station connected, AID=%d", event->aid);
                WifiAp_ResetTimeout();
                break;
            }
            case WIFI_EVENT_AP_STADISCONNECTED: {
                wifi_event_ap_stadisconnected_t *event = (wifi_event_ap_stadisconnected_t *)event_data;
                ESP_LOGI(TAG, "Station disconnected, AID=%d", event->aid);
                break;
            }
            default:
                break;
        }
    }
}

esp_err_t WifiAp_Init(void)
{
    if (s_initialized) return ESP_OK;

    // 创建 AP netif
    s_ap_netif = esp_netif_create_default_wifi_ap();
    if (!s_ap_netif) {
        ESP_LOGE(TAG, "Failed to create AP netif");
        return ESP_FAIL;
    }

    // 注册事件
    esp_event_handler_instance_t instance_any_id;
    ESP_ERROR_CHECK(esp_event_handler_instance_register(
        WIFI_EVENT, ESP_EVENT_ANY_ID, &wifi_event_handler, NULL, &instance_any_id));

    // 创建超时定时器
    const esp_timer_create_args_t timer_args = {
        .callback = timeout_timer_cb,
        .name = "wifi_ap_timeout"
    };
    ESP_ERROR_CHECK(esp_timer_create(&timer_args, &s_timeout_timer));

    s_initialized = true;
    ESP_LOGI(TAG, "Init done");
    return ESP_OK;
}

esp_err_t WifiAp_Start(void)
{
    if (!s_initialized) {
        ESP_LOGE(TAG, "Not initialized");
        return ESP_ERR_INVALID_STATE;
    }

    if (s_state == WIFI_AP_STATE_ACTIVE) {
        ESP_LOGW(TAG, "Already active");
        return ESP_OK;
    }

    wifi_config_t wifi_config = {};
    strncpy((char *)wifi_config.ap.ssid, WIFI_AP_SSID, sizeof(wifi_config.ap.ssid) - 1);
    wifi_config.ap.ssid_len = strlen(WIFI_AP_SSID);
    wifi_config.ap.channel = WIFI_AP_CHANNEL;
    wifi_config.ap.max_connection = WIFI_AP_MAX_CONN;
    wifi_config.ap.authmode = WIFI_AUTH_OPEN;  // 无密码

    ESP_ERROR_CHECK(esp_wifi_set_mode(WIFI_MODE_APSTA));
    ESP_ERROR_CHECK(esp_wifi_set_config(WIFI_IF_AP, &wifi_config));
    ESP_ERROR_CHECK(esp_wifi_start());

    s_state = WIFI_AP_STATE_ACTIVE;
    WifiAp_ResetTimeout();

    if (s_state_cb) s_state_cb(s_state);

    ESP_LOGI(TAG, "AP started: %s", WIFI_AP_SSID);
    return ESP_OK;
}

esp_err_t WifiAp_Stop(void)
{
    if (s_state == WIFI_AP_STATE_IDLE) return ESP_OK;

    if (s_timeout_timer) {
        esp_timer_stop(s_timeout_timer);
    }

    esp_wifi_stop();
    s_state = WIFI_AP_STATE_IDLE;

    if (s_state_cb) s_state_cb(s_state);

    ESP_LOGI(TAG, "AP stopped");
    return ESP_OK;
}

bool WifiAp_IsActive(void)
{
    return s_state == WIFI_AP_STATE_ACTIVE;
}

void WifiAp_ResetTimeout(void)
{
    if (s_timeout_timer && s_state == WIFI_AP_STATE_ACTIVE) {
        esp_timer_stop(s_timeout_timer);
        esp_timer_start_once(s_timeout_timer, WIFI_AP_TIMEOUT_MS * 1000);
    }
}

void WifiAp_SetStateCallback(wifi_ap_state_cb_t cb)
{
    s_state_cb = cb;
}
```

```cpp
// 响应 helper
static esp_err_t send_json(httpd_req_t *req, cJSON *json, int status_code)
{
    char *str = cJSON_PrintUnformatted(json);
    cJSON_Delete(json);
    if (!str) {
        httpd_resp_send_err(req, HTTPD_500_INTERNAL_SERVER_ERROR, "JSON error");
        return ESP_FAIL;
    }

    httpd_resp_set_type(req, "application/json");
    httpd_resp_set_hdr(req, "Access-Control-Allow-Origin", "*");
    httpd_resp_set_status(req, status_code == 200 ? "200 OK" : "400 Bad Request");
    httpd_resp_send(req, str, strlen(str));
    free(str);
    return ESP_OK;
}

static esp_err_t send_ok(httpd_req_t *req, cJSON *data)
{
    cJSON *resp = cJSON_CreateObject();
    cJSON_AddBoolToObject(resp, "ok", true);
    if (data) cJSON_AddItemToObject(resp, "data", data);
    return send_json(req, resp, 200);
}

static esp_err_t send_error(httpd_req_t *req, const char *msg, int code)
{
    cJSON *resp = cJSON_CreateObject();
    cJSON_AddBoolToObject(resp, "ok", false);
    cJSON_AddStringToObject(resp, "error", msg);
    return send_json(req, resp, code);
}

// 解析请求体 JSON
static cJSON *parse_body(httpd_req_t *req)
{
    if (req->content_len <= 0 || req->content_len > 2048) return NULL;

    char *buf = (char *)malloc(req->content_len + 1);
    if (!buf) return NULL;

    int ret = httpd_req_recv(req, buf, req->content_len);
    if (ret <= 0) {
        free(buf);
        return NULL;
    }
    buf[ret] = '\0';

    cJSON *json = cJSON_Parse(buf);
    free(buf);
    return json;
}

// 从 URI 提取 ID: /api/lists/l_001 → "l_001"
static bool extract_id(const char *uri, const char *prefix, char *out, int out_len)
{
    const char *start = strstr(uri, prefix);
    if (!start) return false;
    start += strlen(prefix);

    const char *end = strchr(start, '/');
    if (!end) end = start + strlen(start);

    int len = end - start;
    if (len <= 0 || len >= out_len) return false;

    strncpy(out, start, len);
    out[len] = '\0';
    return true;
}

// GET /api/lists
static esp_err_t handle_get_lists(httpd_req_t *req)
{
    todo_list_t lists[TODO_MAX_LISTS];
    uint8_t count = 0;
    TodoMgr_GetLists(lists, &count);

    cJSON *arr = cJSON_CreateArray();
    for (int i = 0; i < count; i++) {
        cJSON *item = cJSON_CreateObject();
        cJSON_AddStringToObject(item, "id", lists[i].id);
        cJSON_AddStringToObject(item, "name", lists[i].name);
        cJSON_AddStringToObject(item, "icon", lists[i].icon);
        cJSON_AddNumberToObject(item, "order", lists[i].order);
        cJSON_AddNumberToObject(item, "task_count", lists[i].task_count);

        // 计算已完成数
        todo_task_t tasks[TODO_MAX_TASKS];
        uint16_t task_count = 0;
        TodoMgr_GetTasks(lists[i].id, tasks, &task_count);
        uint16_t done_count = 0;
        for (int j = 0; j < task_count; j++) {
            if (tasks[j].done) done_count++;
        }
        cJSON_AddNumberToObject(item, "done_count", done_count);

        cJSON_AddItemToArray(arr, item);
    }

    return send_ok(req, arr);

    httpd_uri_t uri_create_task = { .uri = "/api/lists/*/tasks", .method = HTTP_POST, .handler = handle_create_task };
    httpd_uri_t uri_update_task = { .uri = "/api/tasks/*", .method = HTTP_PUT, .handler = handle_update_task };
    httpd_uri_t uri_delete_task = { .uri = "/api/tasks/*", .method = HTTP_DELETE, .handler = handle_delete_task };
    httpd_uri_t uri_options = { .uri = "/*", .method = HTTP_OPTIONS, .handler = handle_options };

    httpd_register_uri_handler(s_server, &uri_get_lists);
    httpd_register_uri_handler(s_server, &uri_create_list);
    httpd_register_uri_handler(s_server, &uri_update_list);
    httpd_register_uri_handler(s_server, &uri_delete_list);
    httpd_register_uri_handler(s_server, &uri_get_tasks);
    httpd_register_uri_handler(s_server, &uri_create_task);
    httpd_register_uri_handler(s_server, &uri_update_task);
    httpd_register_uri_handler(s_server, &uri_delete_task);
    httpd_register_uri_handler(s_server, &uri_options);

    ESP_LOGI(TAG, "HTTP server started");
    return ESP_OK;
}

esp_err_t TodoWeb_Stop(void)
{
    if (s_server) {
        httpd_stop(s_server);
        s_server = NULL;
        ESP_LOGI(TAG, "HTTP server stopped");
    }
    return ESP_OK;
}

bool TodoWeb_IsRunning(void)
{
    return s_server != NULL;
}
```

```html
const API = '';
let lists = [];
let currentListId = null;

async function loadLists() {
  const res = await fetch(API + '/api/lists');
  const data = await res.json();
  if (data.ok) {
    lists = data.data;
    if (!currentListId && lists.length > 0) currentListId = lists[0].id;
    renderTabs();
    loadTasks();
  }
}

async function loadTasks() {
  if (!currentListId) {
    document.getElementById('taskList').innerHTML = '<p style="padding:16px;color:#999">暂无清单，请先创建</p>';
    return;
  }
  const res = await fetch(API + '/api/lists/' + currentListId + '/tasks');
  const data = await res.json();
  if (data.ok) renderTasks(data.data);
}

function renderTabs() {
  const tabs = document.getElementById('tabs');
  tabs.innerHTML = lists.map(l =>
    `<div class="tab ${l.id === currentListId ? 'active' : ''}" onclick="switchList('${l.id}')">${l.name}</div>`
  ).join('') + '<div class="tab tab-add" onclick="openListModal()">+</div>';
}

function renderTasks(tasks) {
  const list = document.getElementById('taskList');
  if (tasks.length === 0) {
    list.innerHTML = '<p style="padding:16px;color:#999">暂无任务</p>';
    return;
  }
  list.innerHTML = tasks.map(t => {
    const pri = ['', '🟡重要', '🔴紧急'][t.priority] || '';
    const due = t.due ? `📅 ${t.due}` : '';
    return `<div class="task ${t.done ? 'done' : ''}">
      <input type="checkbox" class="task-checkbox" ${t.done ? 'checked' : ''} onchange="toggleTask('${t.id}', this.checked)">
      <div class="task-content">
        <div class="task-text ${t.done ? 'done' : ''}">${t.text}</div>
        <div class="task-meta">${pri} ${due}</div>
      </div>
      <span class="task-delete" onclick="deleteTask('${t.id}')">×</span>
    </div>`;
  }).join('');
}

function switchList(id) {
  currentListId = id;
  loadLists();
}

async function toggleTask(id, done) {
  await fetch(API + '/api/tasks/' + id, {
    method: 'PUT', headers: {'Content-Type':'application/json'},
    body: JSON.stringify({done})
  });
  loadTasks();
}

async function deleteTask(id) {
  if (!confirm('删除此任务？')) return;
  await fetch(API + '/api/tasks/' + id, {method: 'DELETE'});
  loadTasks();
}

function openModal() { document.getElementById('taskModal').classList.add('active'); }

function openListModal() { document.getElementById('listModal').classList.add('active'); }

async function submitTask() {
  const text = document.getElementById('taskText').value.trim();
  if (!text) return alert('请输入任务内容');
  const priority = parseInt(document.getElementById('taskPriority').value);

  loadTasks();
}
```

### Step 2: 创建 scripts/build_web.sh

```bash
#!/bin/bash
set -e

SCRIPT_DIR="$(cd "$(dirname "$0")" && pwd)"
PROJECT_DIR="$(dirname "$SCRIPT_DIR")"
HTML_FILE="$PROJECT_DIR/web_dev/index.html"
OUTPUT="$PROJECT_DIR/components/todo_web/web_page.h"

if [ ! -f "$HTML_FILE" ]; then
  echo "Error: $HTML_FILE not found"
  exit 1
fi

# gzip 压缩
gzip -c "$HTML_FILE" > /tmp/web_page.gz

# 生成 C 数组
xxd -i /tmp/web_page.gz > "$OUTPUT"

# 添加头文件保护
sed -i '1i\
#ifndef WEB_PAGE_H\
#define WEB_PAGE_H\
\
#include <stdint.h>\
\
' "$OUTPUT"

echo 'extern const uint8_t web_page_gz[];' >> "$OUTPUT"
echo 'extern const unsigned int web_page_gz_len;' >> "$OUTPUT"
echo '#endif' >> "$OUTPUT"

# 清理变量名
sed -i 's/web_page_gz/const uint8_t web_page_gz[]/g' "$OUTPUT"
sed -i 's/web_page_gz_len/const unsigned int web_page_gz_len/g' "$OUTPUT"

rm -f /tmp/web_page.gz

echo "Generated $OUTPUT ($(wc -c < "$HTML_FILE") bytes → $(wc -c < /tmp/web_page.gz 2>/dev/null || echo '?') bytes gzip)"
```

```cpp
#include "todo_ui.h"
#include "todo_manager.h"

static lv_obj_t *s_parent = NULL;
static lv_obj_t *s_lists_page = NULL;

// 状态栏

static void list_click_cb(lv_event_t *e)
{
    const char *id = (const char *)lv_event_get_user_data(e);
    TodoUi_ShowPage(TODO_PAGE_TASKS, id);
}

static void task_check_cb(lv_event_t *e)
{
    const char *task_id = (const char *)lv_event_get_user_data(e);
    lv_obj_t *cb = lv_event_get_target(e);
    bool done = lv_obj_has_state(cb, LV_STATE_CHECKED);
    TodoMgr_UpdateTask(task_id, &done, NULL, NULL);
}

static void refresh_lists_page(void)
{
    // 清空
    lv_obj_clean(s_list_container);

    todo_list_t lists[TODO_MAX_LISTS];
    uint8_t count = 0;
    TodoMgr_GetLists(lists, &count);

    for (int i = 0; i < count; i++) {
        char buf[64];
        snprintf(buf, sizeof(buf), "%s %s  %d/%d",
                 lists[i].icon, lists[i].name,
                 lists[i].task_count > 0 ? lists[i].task_count - 0 : 0,
                 lists[i].task_count);

        lv_obj_t *btn = lv_btn_create(s_list_container);
        lv_obj_set_size(btn, 170, 30);
        lv_obj_set_style_bg_color(btn, lv_color_white(), 0);
        lv_obj_set_style_border_width(btn, 1, 0);
        lv_obj_add_event_cb(btn, list_click_cb, LV_EVENT_CLICKED, (void *)lists[i].id);

        lv_obj_t *lbl = lv_label_create(btn);
        lv_label_set_text(lbl, buf);
        lv_obj_set_style_text_color(lbl, lv_color_black(), 0);
        lv_obj_set_style_text_font(lbl, &lv_font_montserrat_12, 0);
        lv_obj_center(lbl);
    }
}

static void refresh_tasks_page(void)
{
    if (s_current_list_id[0] == '\0') return;

    lv_obj_clean(s_task_container);

    // 更新标题
    todo_list_t lists[TODO_MAX_LISTS];
    uint8_t list_count = 0;
    TodoMgr_GetLists(lists, &list_count);

    for (int i = 0; i < list_count; i++) {
        if (strcmp(lists[i].id, s_current_list_id) == 0) {
            char title[48];
            snprintf(title, sizeof(title), "%s %s", lists[i].icon, lists[i].name);
            lv_label_set_text(s_task_title, title);
            break;
        }
    }

    todo_task_t tasks[TODO_MAX_TASKS];
    uint16_t task_count = 0;
    TodoMgr_GetTasks(s_current_list_id, tasks, &task_count);

    for (int i = 0; i < task_count; i++) {
        lv_obj_t *cb = lv_checkbox_create(s_task_container);
        char label[80];
        const char *pri = "";
        if (tasks[i].priority == 1) pri = " [!]";
        else if (tasks[i].priority == 2) pri = " [!!]";

        snprintf(label, sizeof(label), "%s%s%s%s",
                 tasks[i].text,
                 pri,
                 tasks[i].due[0] ? " " : "",
                 tasks[i].due);
        lv_checkbox_set_text(cb, label);

        if (tasks[i].done) {
            lv_obj_add_state(cb, LV_STATE_CHECKED);
            lv_obj_set_style_text_color(cb, lv_color_hex(0x999999), 0);
        }

        lv_obj_add_event_cb(cb, task_check_cb, LV_EVENT_VALUE_CHANGED, (void *)tasks[i].id);
    }
}

esp_err_t TodoUi_Init(lv_obj_t *parent)
{
    s_parent = parent;

    create_status_bar(parent);
    create_lists_page(parent);
    create_tasks_page(parent);

    // 初始显示清单页
    lv_obj_clear_flag(s_lists_page, LV_OBJ_FLAG_HIDDEN);
    lv_obj_add_flag(s_tasks_page, LV_OBJ_FLAG_HIDDEN);

    refresh_lists_page();

    return ESP_OK;
}

void TodoUi_Refresh(void)
{
    if (s_current_page == TODO_PAGE_LISTS) {
        refresh_lists_page();
    } else {
        refresh_tasks_page();
    }
}

void TodoUi_ShowPage(todo_page_type_t page, const char *list_id)
{
    s_current_page = page;

    if (page == TODO_PAGE_LISTS) {
        lv_obj_clear_flag(s_lists_page, LV_OBJ_FLAG_HIDDEN);
        lv_obj_add_flag(s_tasks_page, LV_OBJ_FLAG_HIDDEN);
        refresh_lists_page();
    } else {
        lv_obj_add_flag(s_lists_page, LV_OBJ_FLAG_HIDDEN);
        lv_obj_clear_flag(s_tasks_page, LV_OBJ_FLAG_HIDDEN);
        if (list_id) {
            strncpy(s_current_list_id, list_id, sizeof(s_current_list_id) - 1);
        }
        refresh_tasks_page();
    }
}

todo_page_type_t TodoUi_GetCurrentPage(void)
{
    return s_current_page;
}
```

```cpp
/* BOOT key long-press detection */
static void boot_key_task(void *arg)
{
    uint32_t press_start = 0;

    for (;;) {
        if (gpio_get_level(0) == 0) {  // BOOT pressed (low active)
            if (press_start == 0) {
                press_start = xTaskGetTickCount() * portTICK_PERIOD_MS;
            } else if ((xTaskGetTickCount() * portTICK_PERIOD_MS - press_start) > 2000) {
                // Long press 2 seconds
                ESP_LOGI(TAG, "BOOT long press detected");

                if (WifiAp_IsActive()) {
                    TodoWeb_Stop();
                    WifiAp_Stop();
                    ESP_LOGI(TAG, "AP stopped");
                } else {
                    if (WifiAp_Start() == ESP_OK) {
                        TodoWeb_Start();
                        ESP_LOGI(TAG, "AP started, connect to 'AIoT-TODO'");
                    }
                }

                press_start = 0;
                vTaskDelay(pdMS_TO_TICKS(500));  // Debounce
            }
        } else {
            press_start = 0;
        }
        vTaskDelay(pdMS_TO_TICKS(50));
    }
}
```
